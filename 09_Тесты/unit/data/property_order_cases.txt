# Случаи единого порядка свойств (№23). Общие для юнит-теста PropertyOrderTests (C#) и статического теста оракула
# oracles.canonical (Python). Имена через «|», пробелы внутри имени значимы. Мастер-список — словарь по умолчанию.
# canonical — порядок уровня после наведения, move — что по одному уходит в конец.
# «Примечание», «Формат», «Раздел» — всегда последние, в этом порядке: так их оставляет «Применить» MProp
# (FrmMProp 3021–3053), и MProp с надстройкой не переставляют их друг за другом (решение владельца 24.09.2026).

# уже в каноне
current=Обозначение|Наименование|Масса|Материал|Формат
canonical=Обозначение|Наименование|Масса|Материал|Формат
move=

# шаблон детали до 24.09.2026: «Формат» уходит в конец
current=Обозначение|Наименование|Формат|Масса|Материал
canonical=Обозначение|Наименование|Масса|Материал|Формат
move=Формат

# своё свойство между известными уходит в конец
current=Обозначение|Моё|Наименование
canonical=Обозначение|Наименование|Моё
move=Моё

# свои свойства сохраняют свой порядок
current=Б_моё|Обозначение|А_моё
canonical=Обозначение|Б_моё|А_моё
move=Б_моё|А_моё

# исполнение после «Применить» MProp: «Примечание», «Формат», «Раздел» дописаны в конец — там им и место
current=Обозначение|Масса_ФБ|Масса_Таблица|Материал_ФБ|Материал_Таблица|Примечание|Формат|Раздел
canonical=Обозначение|Масса_ФБ|Масса_Таблица|Материал_ФБ|Материал_Таблица|Примечание|Формат|Раздел
move=

# после MProp со своими свойствами: свои — перед «Примечание», «Формат», «Раздел»
current=Обозначение|Масса_ФБ|Моё|Примечание|Формат|Раздел
canonical=Обозначение|Масса_ФБ|Моё|Примечание|Формат|Раздел
move=

# прежний порядок (до 24.09.2026, по строкам словаря): три переноса, один раз
current=Обозначение|Формат|Примечание|Раздел|Масса_ФБ|Моё
canonical=Обозначение|Масса_ФБ|Моё|Примечание|Формат|Раздел
move=Примечание|Формат|Раздел

# «Примечание» перед «Формат», как пишет MProp, а не как строки словаря (7 — «Формат», 8 — «Примечание»)
current=Обозначение|Формат|Примечание
canonical=Обозначение|Примечание|Формат
move=Формат

# первое «Применить» MProp дописывает свои служебные после «Раздел» — один раз переносятся они
current=Обозначение|Масса_ФБ|Примечание|Формат|Раздел|Сборка1_ФБ|Проверил
canonical=Обозначение|Сборка1_ФБ|Проверил|Масса_ФБ|Примечание|Формат|Раздел
move=Масса_ФБ|Примечание|Формат|Раздел

# «масса» и «Обозначение » с пробелом — не словарные, это свои свойства конструктора
current=Обозначение |масса|Обозначение
canonical=Обозначение|Обозначение |масса
move=Обозначение |масса

# пустой уровень
current=

move=

# уровень из одного своего свойства
current=Моё
canonical=Моё
move=

# «Операции», дописанные кнопкой ЛЗК в конец, уже на своём месте
current=Обозначение|Наименование|Масса_ФБ|Операции
canonical=Обозначение|Наименование|Масса_ФБ|Операции
move=

# «Операции» после своих свойств — переносится своё, а не «Операции»
current=Обозначение|Наименование|Моё|Операции
canonical=Обозначение|Наименование|Операции|Моё

# служебные SWPlus и прежние v5 — после словаря и имён надстройки
