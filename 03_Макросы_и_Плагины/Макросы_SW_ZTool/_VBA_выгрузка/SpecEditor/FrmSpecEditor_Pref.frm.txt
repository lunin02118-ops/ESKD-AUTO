Private Sub UserForm_Activate()
HWNDActiveWindow
FrmSpecEditor.MyProperties
If prpTestStandard = 1 Then
    FrmSpecEditor.MyStandard
    TxtFontName.Value = stdFontName
    TxtFontSize.Value = LTrim(CStr(stdFontSize))
    If stdFontItalic = 1 Then
        ChkFontItalic.Value = True
    Else
        ChkFontItalic.Value = False
    End If
    If stdFontBold = 1 Then
        ChkFontBold.Value = True
    Else
        ChkFontBold.Value = False
    End If
    TxtFontName.Enabled = False
    TxtFontSize.Enabled = False
    ChkFontItalic.Enabled = False
    ChkFontBold.Enabled = False
Else
    TxtFontName.Enabled = True
    TxtFontSize.Enabled = True
    ChkFontItalic.Enabled = True
    ChkFontBold.Enabled = True
End If
' Чтение ini файла
Open sSource4 For Input As #1
    If prpTestStandard = 1 Then
        Line Input #1, strTemp
        Line Input #1, strTemp
        Line Input #1, strTemp
        Line Input #1, strTemp
    Else
        Line Input #1, strTemp ' Шрифт
        TxtFontName.Value = strTemp
        Line Input #1, strTemp ' Размер шрифта
        strTemp = Replace(strTemp, ".", ",")
        TxtFontSize.Value = strTemp
        Line Input #1, strTemp ' Наклонный
        If strTemp = "1" Then
            ChkFontItalic.Value = True
        Else
            ChkFontItalic.Value = False
        End If
        Line Input #1, strTemp ' Жирный
        If strTemp = "1" Then
            ChkFontBold.Value = True
        Else
            ChkFontBold.Value = False
        End If
    End If
    Line Input #1, strTemp
    TxtFontWidth.Value = strTemp ' Сжатие общее
    Line Input #1, strTemp
    TxtRowWidth.Value = strTemp ' Сжатие длинных строк
    Line Input #1, strTemp
    TxtRemarkWidth.Value = strTemp ' Сжатие столбца Примечание
    '
    Line Input #1, strTemp ' Обозначение
    If strTemp = "1" Then
        ChkNumber.Value = True
    Else
        ChkNumber.Value = False
    End If
    Line Input #1, strTemp ' Наименование
    If strTemp = "1" Then
        ChkDescription.Value = True
    Else
        ChkDescription.Value = False
    End If
    Line Input #1, strTemp
    TxtSP1.Value = strTemp
    Line Input #1, strTemp
    TxtSP2.Value = strTemp
    Line Input #1, strTemp
    TxtGSP1.Value = strTemp
    Line Input #1, strTemp
    TxtGSP2.Value = strTemp
    Line Input #1, strTemp
    TxtVP1.Value = strTemp
    Line Input #1, strTemp
    TxtVP2.Value = strTemp
    '
    ' Резервирование строк
    Line Input #1, strTemp ' После каждой строки
    If strTemp = "1" Then
        ChkLine.Value = True
    Else
        ChkLine.Value = False
    End If
    Line Input #1, strTemp
    TxtLine.Value = strTemp
    '
    Line Input #1, strTemp ' В конце раздела
    If strTemp = "1" Then
        ChkSection.Value = True
    Else
        ChkSection.Value = False
    End If
    Line Input #1, strTemp
    TxtSection.Value = strTemp
    '
    ' Резервирование позиций
    Line Input #1, strTemp ' После каждой строки
    If strTemp = "1" Then
        ChkPosLine.Value = True
    Else
        ChkPosLine.Value = False
    End If
    Line Input #1, strTemp
    TxtPosLine.Value = strTemp
    '
    Line Input #1, strTemp ' В конце раздела
    If strTemp = "1" Then
        ChkPosSection.Value = True
    Else
        ChkPosSection.Value = False
    End If
    Line Input #1, strTemp
    TxtPosSection.Value = strTemp
    '
    Line Input #1, strTemp ' По числу резервированных строк
    If strTemp = "1" Then
        ChkPosReserve.Value = True
    Else
        ChkPosReserve.Value = False
    End If
    '
    ' Форматирование
    Line Input #1, strTemp ' Помечать цветом
    If strTemp = "1" Then
        ChkForm0.Value = True
    Else
        ChkForm0.Value = False
    End If
    Line Input #1, strTemp ' Убирать базовую часть обозначения для исполнений
    If strTemp = "1" Then
        ChkForm1.Value = True
    Else
        ChkForm1.Value = False
    End If
    Line Input #1, strTemp ' Группировать стандартные
    If strTemp = "1" Then
        ChkForm2.Value = True
    Else
        ChkForm2.Value = False
    End If
    '
    Line Input #1, strTemp ' Добавлять ЛРИ
    If strTemp = "1" Then
        ChkLRI.Value = True
    Else
        ChkLRI.Value = False
    End If
    '
    Line Input #1, strTemp ' Использовать быструю сортировку
    If strTemp = "1" Then
        ChkSort.Value = True
    Else
        ChkSort.Value = False
    End If
    '
    Line Input #1, strTemp ' Сортировка Прочих как Стандартных
    If strTemp = "1" Then
        ChkOther.Value = True
    Else
        ChkOther.Value = False
    End If
Close #1
ChkDescription_Click
ChkNumber_Click
If prpLeftTopCorner = 1 Then ' Окно макроса в левом верхнем углу
    FrmSpecEditor_Pref.Left = 10
    FrmSpecEditor_Pref.Top = 10
End If
If prpTopAll = 1 Then ' Окно макроса поверх всех
    KeepFormOnTop
End If
End Sub
