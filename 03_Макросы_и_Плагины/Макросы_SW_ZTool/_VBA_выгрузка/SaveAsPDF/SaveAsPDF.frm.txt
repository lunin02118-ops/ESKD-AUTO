Private Sub UserForm_Activate()
HWNDActiveWindow
If prpLeftTopCorner = 1 Then ' Окно макроса в левом верхнем углу
    SaveAsPDF.Left = 10
    SaveAsPDF.Top = 10
End If
If prpTopAll = 1 Then ' Окно макроса поверх всех
    KeepFormOnTop
End If
Set swApp = Application.SldWorks
'Set swApp = CreateObject("SldWorks.Application")
Set swModel = swApp.ActiveDoc
Set swModelDocExt = swModel.Extension
Set fs = CreateObject("Scripting.FileSystemObject")

' Проверка открытого документа
If swModel Is Nothing Then
    swApp.SendMsgToUser ("Откройте чертеж! Файл не создан! Свойства не внесены!")
    Finish
    Exit Sub
End If
If swModel.GetType <> swDocDRAWING Then
    swApp.SendMsgToUser ("Откройте чертеж! Файл не создан! Свойства не внесены!")
    Finish
    Exit Sub
End If
Set swDraw = swModel

' Проверка пути сохранения
sPathName = swDraw.GetPathName
If sPathName = "" Then
    swApp.SendMsgToUser ("Сохраните чертеж! Файл не создан! Свойства не внесены!")
    Finish
    Exit Sub
End If

' Имя чертежа и имена листов
sDrawName = swDraw.GetTitle
vSheetNames = swDraw.GetSheetNames
vSheetNamesPrint = swDraw.GetSheetNames
' Отрезаем лишнее от имени файла чертежа
varTemp = InStrRev(sDrawName, "-")
If varTemp > 0 Then
    sDrawName = Left$(sDrawName, varTemp - 2)
Else
    swApp.SendMsgToUser ("Неверное имя чертежа! PDF не создан! Свойства не внесены!")
    Finish
    Exit Sub
End If
sDirName = Left$(sPathName, Len(sPathName) - Len(sDrawName) - 7)

Debug.Print "sPathName=", sPathName
Debug.Print "sDrawName=", sDrawName
Debug.Print "sDirName=", sDirName

' Пути
Source = swApp.GetCurrentMacroPathName ' Get macro path+filename
Source1 = Left$(Source, Len(Source) - 23) & "SpecEditor\MyProperties_1.ini" ' Путь к списку свойств
Source2 = Left$(Source, Len(Source) - 3) & "ini" ' Путь к ini-файлу
Source3 = Left$(Source, Len(Source) - 23) & "SpecEditor\MyProperties_2.ini" ' Путь к списку свойств
Source4 = Left$(Source, Len(Source) - 23) & "SpecEditor\MyStandard.sldstd" ' Путь к файлу стандарта
Source5 = Left$(Source, Len(Source) - 13) & "PDFCreator.swp" ' Путь к макросу PDFCreator.swp
Source6 = Left$(Source, Len(Source) - 13) & "PDFCreator.dat" ' Путь к файлу PDFCreator.dat

' Чтение ini файла
Open Source2 For Input As #1
    Line Input #1, strTemp
    sIni1 = CInt(Left$(strTemp, 1)) ' 0 - SaveAs PDF; 1 - PDFCreator
    If sIni1 = 1 Then
        If MPrint = 0 Then
            lRetval = swApp.SendMsgToUser2("PDFCreator не установлен. Будет использован встроенный PDF. Продолжить?", swMbWarning, swMbYesNo)
            If lRetval = swMbHitNo Then
                swApp.SendMsgToUser ("PDF не создан! Свойства не внесены!")
                Close #1
                Finish
                Exit Sub
            End If
            sIni1 = 0
        End If
    End If
    Line Input #1, strTemp
    sIni2 = CInt(Left$(strTemp, 1)) ' 0 - Independent mode; 1 - Part of the macros group
    Line Input #1, strTemp
    sIni3 = CInt(Left$(strTemp, 1)) ' 0 - PDF; 1 - PNG, 2 = JPEG, 3 = BMP, 4 = PCX, 5 = TIFF
    Line Input #1, strTemp
    sIni4 = CInt(Left$(strTemp, 1)) ' 0 - Рядом; 1 - Папка PDF или TIFF
    Line Input #1, strTemp
    sIni5 = CInt(Left$(strTemp, 1)) ' 0 - Не учитывать Изм.; 1 - Учитывать Изм.
    Line Input #1, strTemp
    sIni6 = CInt(Left$(strTemp, 1)) ' 0 - Рядом; 1 - Папка Архив (мода fd)
    Line Input #1, strTemp
    varTemp = InStr(strTemp, "'")
    sIni7 = Left$(strTemp, varTemp - 2) ' Имя папки
    Line Input #1, strTemp
    varTemp = InStr(strTemp, "'")
    sIni8 = Left$(strTemp, varTemp - 2) ' Часть суффикса
    Line Input #1, strTemp
    varTemp = InStr(strTemp, "'")
    sIni9 = Left$(strTemp, varTemp - 2) ' Имя папки Архив (мода fd)
    Line Input #1, strTemp
    sIni10 = CInt(Left$(strTemp, 1)) ' 0 - ч/б; 1 - цвет.
    Line Input #1, strTemp
    varTemp = InStr(strTemp, "'")
    sIni11 = CInt(Left$(strTemp, varTemp - 2)) ' DPI
Close #1

' Определение имени файла
sPathName = Left(sPathName, Len(sPathName) - 7)
If sIni3 = 0 Then
    sPathName = sPathName & ".pdf"
Else
    sPathName = sPathName & ".tif"
End If
' Добавляем путь к папке
If sIni4 = 1 Then
    varTemp = InStrRev(sPathName, "\")
    If sIni3 = 0 Then ' PDF
        sPathName = Left$(sPathName, varTemp - 1) & "\PDF" & Right$(sPathName, Len(sPathName) - varTemp + 1)
        sDirName = sDirName & "PDF\"
    Else ' TIF
        sPathName = Left$(sPathName, varTemp - 1) & "\TIF" & Right$(sPathName, Len(sPathName) - varTemp + 1)
        sDirName = sDirName & "TIF\"
    End If
    If fs.FolderExists(sDirName) = False Then
        MkDir sDirName
    End If
End If

m = 0 ' Метка проблем с моделью
If sIni2 = 1 Then
    ' Получение первого листа
    ok = swDraw.ActivateSheet(vSheetNames(0))
    Set swSheet = swDraw.GetCurrentSheet
    
    ' Проверка наличия модели
    Set swView = swDraw.GetFirstView
    m1 = 0
    If swSheet.CustomPropertyView = "По умолчанию" Or swSheet.CustomPropertyView = "Default" Then
        Set swView = swView.GetNextView ' Получаем первый вид
    Else
        Do Until swView Is Nothing
            If swView.GetName2 = swSheet.CustomPropertyView Then
                m1 = 1
                Exit Do
            End If
            Set swView = swView.GetNextView
        Loop
        If m1 = 0 Then
            Set swView = swDraw.GetFirstView
            Set swView = swView.GetNextView
            swApp.SendMsgToUser ("Не удалось определить вид из свойств листа. Используется первый вид.")
        End If
    End If
    If swView Is Nothing Then
        lRetval = swApp.SendMsgToUser2("Отсутствует модель! Продолжить?", swMbWarning, swMbYesNo)
        If lRetval = swMbHitNo Then
            swApp.SendMsgToUser ("PDF не создан! Свойства не внесены!")
            Finish
            Exit Sub
        Else
            m = 1
        End If
    Else
        If swView.ReferencedDocument Is Nothing Then
            lRetval = swApp.SendMsgToUser2("Отсутствует модель! Продолжить?", swMbWarning, swMbYesNo)
            If lRetval = swMbHitNo Then
                swApp.SendMsgToUser ("PDF не создан! Свойства не внесены!")
                Finish
                Exit Sub
            Else
                m = 1
            End If
        End If
    End If
End If
sDelName = ""
MRev = 0 ' Метка Изм.
If sIni2 = 1 And m = 0 Then ' Часть группы макросов и проблем с моделью нет
    MyProperties
    If prpTestStandard = 1 Then
        MyStandard
    End If
    
    ' Получаем имя модели и конфигурации
    Set swModel = swView.ReferencedDocument
    sConfigName = swView.ReferencedConfiguration
    ConfNumb = swModel.GetConfigurationCount ' Общее число конфигураций
    sModelName = swModel.GetTitle
    
    ' Проверка на наличие расширения в имени файла модели
    If Len(sModelName) > 7 Then
        strTemp = Mid$(sModelName, Len(sModelName) - 6, 4)
        If strTemp = ".SLD" Or strTemp = ".sld" Then
            sModelName = Left$(sModelName, Len(sModelName) - 7)
        End If
    End If
    'Debug.Print "sModelName=", sModelName
    'Debug.Print "sConfigName=", sConfigName
    'Debug.Print "Number=", swModel.CustomInfo2("", "Number")
    'Debug.Print "Обозначение=", swModel.CustomInfo2(sConfigName, prpNumber)
    'Debug.Print "Сборка1_ФБ=", swModel.CustomInfo2(sConfigName, prpDocCode)

    ' Проверка правильности имени чертежа
    ' Определяем Number
    sNumber = swModel.CustomInfo2("", "Number")
    sNumber = Replace(sNumber, "$PRP:" & Chr$(34) & "SW-File Name" & Chr$(34), sModelName)

    If prpTestName = 1 Then
        ' Определяем правильное имя чертежа
        varTemp = InStr(sModelName, prpNameSep)
        If swModel.CustomInfo2(sConfigName, "Исполнение") = "1" Or swModel.CustomInfo2(sConfigName, "Исполнение") = "-1" Or swModel.CustomInfo2(sConfigName, "Исполнение") = "True" Then
            If prpFileName = 1 And varTemp > 0 Then
                sDrawTrueName = Left$(sModelName, varTemp - 1) & "-" & Left$(sConfigName, 2) & Right$(sModelName, Len(sModelName) - varTemp + 1)
            Else
                sDrawTrueName = sModelName & "-" & Left$(sConfigName, 2)
            End If
        ElseIf swModel.CustomInfo2(sConfigName, "Исполнение") = "2" Then
            strTemp = swModel.CustomInfo2(sConfigName, prpNumber)
            varTemp1 = InStrRev(strTemp, "-")
            If varTemp1 > 0 Then
                strTemp = Right$(strTemp, Len(strTemp) - varTemp1)
            Else
                strTemp = ""
            End If
            If prpFileName = 1 And varTemp > 0 Then
                sDrawTrueName = Left$(sModelName, varTemp - 1) & "-" & strTemp & Right$(sModelName, Len(sModelName) - varTemp + 1)
            Else
                sDrawTrueName = sModelName & "-" & strTemp
            End If
        Else
            sDrawTrueName = sModelName
        End If
        varTemp = InStr(sDrawTrueName, prpNameSep)
        m = 0
        If Left$(vSheetNames(0), 2) = "VP" Then
            m = 1
            If prpFileName = 1 And varTemp > 0 Then
                sDrawTrueName = Left$(sDrawTrueName, varTemp - 1) & "ВП" & Right$(sDrawTrueName, Len(sDrawTrueName) - varTemp + 1)
            Else
                sDrawTrueName = sDrawTrueName & "ВП"
            End If
        Else
            If swModel.CustomInfo2(sConfigName, prpDocCode) <> "СБ" And swModel.CustomInfo2(sConfigName, prpDocCode) <> "" Then
                If prpFileName = 1 And varTemp > 0 Then
                    sDrawTrueName = Left$(sDrawTrueName, varTemp - 1) & swModel.CustomInfo2(sConfigName, prpDocCode) & Right$(sDrawTrueName, Len(sDrawTrueName) - varTemp + 1)
                Else
                    sDrawTrueName = sDrawTrueName & swModel.CustomInfo2(sConfigName, prpDocCode)
                End If
            End If
        End If
        ' Собственно проверка
        If sDrawName <> sDrawTrueName Then
            lRetval = swApp.SendMsgToUser2("Неверное имя файла чертежа!" & Chr$(10) & "Правильное имя: " & sDrawTrueName & Chr$(10) & "Продолжить?", swMbWarning, swMbYesNo)
            If lRetval = swMbHitNo Then
                swApp.SendMsgToUser ("PDF не создан! Свойства не внесены!")
                Finish
                Exit Sub
            End If
        End If
    End If
    If MRun = 2 Then ' Назначаем имена по Обозначению
        sDrawName = swModel.CustomInfo2(sConfigName, prpNumber)
        If sIni6 = 1 Then
            sDirName = sDirName & sIni9 & "\"
            If fs.FolderExists(sDirName) = False Then
                MkDir sDirName
            End If
        End If
        If sIni3 = 0 Then ' PDF
            sPathName = sDirName & sDrawName & ".pdf"
        Else
            sPathName = sDirName & sDrawName & ".tif"
        End If
    End If
    If prpTestName = 1 Then
        ' Определяем имя старого файла с добавлением СБ
        varTemp = InStr(sPathName, sDrawName) ' Начало имени файла
        varTemp = InStr(varTemp, sPathName, prpNameSep) ' Положение разделителя
        varTemp1 = InStr(sDrawName, prpNameSep)
        If prpFileName = 1 And varTemp > 0 Then
            sDelName = Left$(sPathName, varTemp - 1) & "СБ" & Right$(sPathName, Len(sPathName) - varTemp + 1)
        Else
            If sIni3 = 0 Then ' PDF
                sDelName = Left$(sPathName, Len(sPathName) - 4) & "СБ.pdf"
            Else
                sDelName = Left$(sPathName, Len(sPathName) - 4) & "СБ.tif"
            End If
        End If
        ' Уточнение имени pdf и удаление pdf-ок с неправильными именами
        If sIni1 = 1 Or sIni1 = 0 And sIni3 = 0 Then ' PDFCreator или PDF
            For i = 0 To UBound(vSheetNames) ' Ищем листы спецификации
                If Left$(vSheetNames(i), 2) = "SP" Then
                    m = 1
                End If
            Next i
            If swModel.CustomInfo2(sConfigName, prpDocCode) = "СБ" And m = 0 Then ' Проверяем наличие файлов со старыми именами (без добавления СБ)
                If fs.FileExists(sPathName) = True Then
                    lRetval = swApp.SendMsgToUser2("Существует ранее сохраненный файл " & sPathName & Chr$(10) & "Удаляем его?", swMbWarning, swMbYesNo)
                    If lRetval = swMbHitYes Then
                        Kill sPathName
                    End If
                End If
                If prpFileName = 1 And varTemp > 0 Then
                    sPathName = Left$(sPathName, varTemp - 1) & "СБ" & Right$(sPathName, Len(sPathName) - varTemp + 1)
                    sDrawName = Left$(sDrawName, varTemp1 - 1) & "СБ" & Right$(sDrawName, Len(sDrawName) - varTemp1 + 1)
                Else
                    If sIni3 = 0 Then ' PDF
                        sPathName = Left$(sPathName, Len(sPathName) - 4) & "СБ.pdf"
                    Else
                        sPathName = Left$(sPathName, Len(sPathName) - 4) & "СБ.tif"
                    End If
                    sDrawName = sDrawName & "СБ"
                End If
            End If
            If m = 1 Then ' Проверяем наличие файлов со старыми именами (с добавлением СБ)
                If fs.FileExists(sDelName) = True Then
                    lRetval = swApp.SendMsgToUser2("Существует ранее сохраненный файл " & sDelName & Chr$(10) & "Удаляем его?", swMbWarning, swMbYesNo)
                    If lRetval = swMbHitYes Then
                        Kill sDelName
                    End If
                End If
            End If
        End If
    End If
    
    ' Проверка Заимствования
    strTemp = swModel.CustomInfo2("", prpBor)
    If strTemp <> "" Then
        strTemp = Left$(strTemp, Len(strTemp) - 1)
        strTemp = Replace(strTemp, "$", ", ")
        strMsg = "Внимание! Вы сохраняете документ, который заимствован в следующих проектах:" _
        & Chr$(10) & strTemp & Chr$(10) & "Вы хотите продолжить?"
        lRetval = swApp.SendMsgToUser2(strMsg, swMbQuestion, swMbYesNo)
        If lRetval = swMbHitNo Then
            Finish
            Exit Sub
        End If
    End If
    
    ' Вызов тестов
    Tests
    If MExit = 1 Then
        Finish
        Exit Sub
    End If

    ' Получение и вставка свойств
    ' Наименование и Description
    swDraw.SummaryInfo(0) = swModel.SummaryInfo(0)
    ok = swDraw.AddCustomInfo2("Description", 30, swModel.SummaryInfo(0))
    swDraw.CustomInfo2("", "Description") = swModel.SummaryInfo(0)
    ' Конструктор
    swDraw.SummaryInfo(2) = swModel.SummaryInfo(2)
    ' Number
    ok = swDraw.AddCustomInfo2("Number", 30, sNumber)
    swDraw.CustomInfo2("", "Number") = sNumber
End If
If sIni2 = 1 Then ' Часть группы макросов
    If MRun = 0 Or MRun = 2 Then ' Обновляем дату
        ' Дата
        ' Определяем дату
        MyDate = Date
        MyDay = Day(MyDate)
        MyMonth = Month(MyDate)
        MyYear = Year(MyDate)
        strData = CStr(MyDay) & "." & CStr(MyMonth) & "." & Right$(CStr(MyYear), 2)
        ' Активируем первый лист
        ok = swDraw.ActivateSheet(vSheetNames(0))
        Set swSheet = swDraw.GetCurrentSheet
        strSheetFormatName = swSheet.GetSheetFormatName()
        Set swSelMgr = swDraw.SelectionManager
        ' Проверяем наличие изменений (Revision)
        strTemp = "Revision4@" & strSheetFormatName
        ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
        If ok = True Then ' Есть заметка Revision4
            Set swNote = swSelMgr.GetSelectedObject2(1)
            strTemp = swNote.GetText()
            If Len(strTemp) < 2 Then ' Изменений еще не было
                ' Устанавливаем дату Разраб.
                For i = 0 To UBound(vSheetNames)
                    ok = swDraw.ActivateSheet(vSheetNames(i))
                    Set swSheet = swDraw.GetCurrentSheet
                    strSheetFormatName = swSheet.GetSheetFormatName()
                    strTemp = "Date@" & strSheetFormatName
                    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
                    If ok = True Then
                        Set swNote = swSelMgr.GetSelectedObject2(1)
                        swNote.SetText strData
                    End If
                Next i
            Else ' Изменения были
                MRev = 1
                ' Устанавливаем дату Изм.
                For i = 0 To UBound(vSheetNames)
                    ok = swDraw.ActivateSheet(vSheetNames(i))
                    Set swSheet = swDraw.GetCurrentSheet
                    strSheetFormatName = swSheet.GetSheetFormatName()
                    strTemp = "Revision4@" & strSheetFormatName
                    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
                    If ok = True Then
                        Set swNote = swSelMgr.GetSelectedObject2(1)
                        swNote.SetText strData
                    End If
                Next i

                                Set swNote = swNote.GetNext

                        Else

                    Wend

                        Set swAnn = swNote.GetAnnotation()

                        If swTextFormat.TypeFaceName <> stdFontName Then
                            m = 1
                        End If
                        If swTextFormat.Italic = True And stdFontItalic = 0 Then
                            m = 1
                        ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                            m = 1
                        End If
                        If swTextFormat.Bold = True And stdFontBold = 0 Then
                            m = 1
                        ElseIf swTextFormat.Bold = False And stdFontBold = 1 Then
                            m = 1
                        End If
                    Else
                        m = 1
                    End If
                Else
                    m = 1
                End If

                    Set swNote = swView.GetFirstNote()
                    If Not swNote Is Nothing Then
                        Set swAnn = swNote.GetAnnotation()
                        Set swTextFormat = swAnn.GetTextFormat(0)
                        If swTextFormat.TypeFaceName <> stdFontName Then
                            m = 1
                        End If
                        If swTextFormat.Italic = True And stdFontItalic = 0 Then
                            m = 1
                        ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                            m = 1
                        End If

                            m = 1

                If swTextFormat.TypeFaceName <> stdFontName Or swTextFormat.CharHeight <> stdFontSize / 1000 Then
                    m = 1
                End If
                If swTextFormat.Italic = True And stdFontItalic = 0 Then
                    m = 1
                ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                    m = 1
                End If
                If swTextFormat.Bold = True And stdFontBold = 0 Then
                    m = 1
                ElseIf swTextFormat.Bold = False And stdFontBold = 1 Then
                    m = 1

                    sSheetsNames2 = sSheetsNames2 & vSheetNames(i) & ", "
                End If
            End If
        End If

                    strMsg = "Лист(ы) " & sSheetsNames3 & " не позволяют определить формат." & Chr$(10) & "Для исправления используйте макрос DProp" & Chr$(10) & "Продолжить?"
                    lRetval = swApp.SendMsgToUser2(strMsg, swMbWarning, swMbYesNo)
                    If lRetval = swMbHitNo Then
                        MExit = 1
                        Exit Sub
                    End If
                End If
            End If
        End If
        ' Очистка Примечания от *)
        strTemp = swModel.CustomInfo2(sConfigName, prpRemark)
        If swModel.CustomInfo2(sConfigName, prpFormat) <> "*)" And Left$(strTemp, 2) = "*)" Then
            ok = swModel.AddCustomInfo3(sConfigName, prpRemark, 30, "")
            swModel.CustomInfo2(sConfigName, prpRemark) = ""
        End If
        ' Синхронизация свойств
        If ConfNumb = 1 Then
            ok = swModel.AddCustomInfo3("", prpRemark, 30, "")
            ok = swModel.AddCustomInfo3("", prpFormat, 30, "")
            swModel.CustomInfo2("", prpRemark) = swModel.CustomInfo2(sConfigName, prpRemark)
            swModel.CustomInfo2("", prpFormat) = swModel.CustomInfo2(sConfigName, prpFormat)
        End If
    End If
End If
End Sub
