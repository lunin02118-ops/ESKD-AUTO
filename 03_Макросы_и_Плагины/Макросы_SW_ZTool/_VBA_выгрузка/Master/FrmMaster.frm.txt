Private Sub CboFormat_DropButtonClick()
CboMulti.Clear
CboMulti.AddItem "нет"
Select Case CboFormat.Text
Case "A4"
    For i = 3 To 9
        CboMulti.AddItem i
    Next i
    Width = 210
    Height = 297
Case "A3"
    For i = 3 To 7
        CboMulti.AddItem i
    Next i
    Width = 297
    Height = 420
Case "A2"
    For i = 3 To 5
        CboMulti.AddItem i
    Next i
    Width = 420
    Height = 594
Case "A1"
    For i = 3 To 4
        CboMulti.AddItem i
    Next i
    Width = 594
    Height = 841
Case "A0"
    For i = 2 To 3
        CboMulti.AddItem i
    Next i
    Width = 841
    Height = 1189
End Select
If CboFormat.Text = "A4" Then
    OptAlbum.Enabled = False
    OptPortrait.Enabled = True
    OptPortrait.Value = True
Else
    OptAlbum.Enabled = True
    OptPortrait.Enabled = True
End If
CboMulti.ListIndex = 0
End Sub
