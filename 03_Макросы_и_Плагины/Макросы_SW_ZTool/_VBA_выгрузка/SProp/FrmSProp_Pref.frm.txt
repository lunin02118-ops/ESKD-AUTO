Private Sub UserForm_Activate()
HWNDActiveWindow
' Чтение ini файла
Open Source6 For Input As #1
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkLight.Value = True
    Else
        ChkLight.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkClean.Value = True
    Else
        ChkClean.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkTest.Value = True
    Else
        ChkTest.Value = False
    End If
Close #1
If prpLeftTopCorner = 1 Then ' Окно макроса в левом верхнем углу
    FrmSProp_Pref.Left = 10
    FrmSProp_Pref.Top = 10
End If
If prpTopAll = 1 Then ' Окно макроса поверх всех
    KeepFormOnTop
End If
End Sub

Private Sub CmdSave_Click()
' Запись в ini
Open Source6 For Output As #1
    If ChkLight.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkClean.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkTest.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
Close #1

End Sub
