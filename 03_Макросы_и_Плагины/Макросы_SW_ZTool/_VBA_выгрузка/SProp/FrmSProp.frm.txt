Private Sub ChkConfig_Click()
If mConfig_Click = 1 Then
    Test
    sStand = swModel.CustomInfo2("", prpNumberDocVP)
    sStand = ReConvert(sStand)
    If ChkConfig.Value = True Then
        If mConfig = 0 Then
            ImgOk.BackColor = &HFF&
        End If
        LblConfig.Visible = True
        LblConfig.Caption = sConfigName
        TxtParam1.Visible = True
        TxtParam.Visible = True
        sName = swModel.CustomInfo2(sConfigName, prpDescription)
        sName = ReConvert(sName)
        If sName = "" Then
            strTemp = swModel.CustomInfo2("", prpDescription)
            TxtPre.Text = ReConvert(strTemp)
            TxtParam1.Text = ""
            TxtParam.Text = ""
        Else
            varTemp = InStr(sName, "$PRP:" & Chr$(34) & "SW-Configuration Name" & Chr$(34))
            If varTemp > 0 Then
                varTemp1 = InStrRev(sName, " ", varTemp)
                If varTemp1 > 0 Then
                    TxtPre.Text = Left$(sName, varTemp1 - 1)
                    Debug.Print Len(sName), varTemp + 28 + Len(sStand) ' 28 символов в $PRP:"SW-Configuration Name"
                    If varTemp > varTemp1 + 1 Then
                        TxtParam1.Text = Mid$(sName, varTemp1 + 1, varTemp - varTemp1 - 1)
                    Else
                        TxtParam1.Text = ""
                    End If
                    If Len(sName) > varTemp + 28 + Len(sStand) Then
                        TxtParam.Text = Mid$(sName, varTemp + 28, Len(sName) - varTemp - 28 - Len(sStand))
                    Else
                        TxtParam.Text = ""
                    End If
                Else
                    TxtPre.Text = Left$(sName, varTemp - 2)
                    TxtParam1.Text = ""
                    TxtParam.Text = ""
                End If
            Else
                TxtPre.Text = sName
                TxtParam1.Text = ""
                TxtParam.Text = ""
            End If
        End If
    Else
        If mConfig = 1 Then
            ImgOk.BackColor = &HFF&
        End If
        LblConfig.Visible = False
        LblConfig.Caption = ""
        TxtParam1.Visible = False
        TxtParam.Visible = False
        TxtParam1.Text = ""
        TxtParam.Text = ""
        sNameVP = swModel.CustomInfo2("", prpDescriptionVP)
        sNameVP = ReConvert(sNameVP)
        TxtPre.Text = sNameVP
        If sNameVP = "" Then
            sName = swModel.CustomInfo2("", prpDescription)
            sName = ReConvert(sName)
            TxtPre.Text = sName
        End If
    End If

    If OptOther.Value = True Or OptStandard.Value = True Then
        If ImgOk.BackColor = &HFF00& Then
            TxtPost.Text = sStand
        Else
            If ChkConfig.Value = True Then
                If varTemp > 0 And Len(sStand) > 0 Then ' Разделитель присутствует
                    TxtPost.Text = Right$(sName, Len(sStand))
                Else
                    TxtPost.Text = sName
                End If
            Else
                TxtPost.Text = sName
            End If
        End If
    Else
        TxtPost.Text = ""
    End If
    Look
End If
End Sub

Private Sub CmdDelete_Click() ' Убиваем все свойства
If MIni2 = 1 Then
    For i = 0 To UBound(vConfNameArr)
        vCustInfoNameArr = swModel.GetCustomInfoNames2(vConfNameArr(i))
        For j = 0 To UBound(vCustInfoNameArr)
            If MDel = 1 And vCustInfoNameArr(j) = prpQuantity Then
            Else
                ok = swModel.DeleteCustomInfo2(vConfNameArr(i), vCustInfoNameArr(j))
            End If
            swModel.CustomInfo2(vConfNameArr(i), "UNIT_OF_MEASURE") = ""
        Next j
    Next i
    vCustInfoNameArr = swModel.GetCustomInfoNames2("")
    For j = 0 To UBound(vCustInfoNameArr)
        If MDel = 1 And vCustInfoNameArr(j) = prpQuantity Then
        Else
            ok = swModel.DeleteCustomInfo2("", vCustInfoNameArr(j))
        End If
    Next j
    swModel.CustomInfo2("", "UNIT_OF_MEASURE") = ""
    swModel.SummaryInfo(swSumInfoTitle) = ""
    swModel.SummaryInfo(swSumInfoSubject) = ""
    swModel.SummaryInfo(swSumInfoAuthor) = ""
    swModel.SummaryInfo(swSumInfoKeywords) = ""
    swModel.SummaryInfo(swSumInfoComment) = ""
End If
End Sub

Private Sub ChkSP_Click()
If ChkSP.Value Then
    strTemp = "Заполните через Файл, Свойства."
    If ChkConfig.Value Then
        If swModel.CustomInfo2(sConfigName, prpQuantity) <> "" Then
            Set swCustPropMgr = swModExt.CustomPropertyManager(sConfigName)
            ok = swCustPropMgr.Get4(prpQuantity, False, strTemp1, strTemp)
        End If
    Else
        If swModel.CustomInfo2("", prpQuantity) <> "" Then
            Set swCustPropMgr = swModExt.CustomPropertyManager("")
            ok = swCustPropMgr.Get4(prpQuantity, False, strTemp1, strTemp)
        End If
    End If
    LblSP.Caption = "Свойство " & Chr$(34) & prpQuantity & Chr$(34) & ":" & Chr$(10) & strTemp
Else
    LblSP.Caption = ""
End If
    'swModel.CustomInfo2("", "UNIT_OF_MEASURE") = " - none - "
End Sub

    ImgOk.Visible = True
Else
    ImgOk.Visible = False
End If

    OptStandard.Value = True
    ChkEM.Value = False
ElseIf Section = "ЭМ-Стандартные изделия" Then
    OptStandard.Value = True
    ChkEM.Value = True
ElseIf Section = "Материалы" Then
    OptMaterial.Value = True
    ChkEM.Value = False
ElseIf Section = "ЭМ-Материалы" Then
    OptMaterial.Value = True
    ChkEM.Value = True

    OptOther.Value = True
    ChkEM.Value = True
Else

End If

    ChkIsFastener.Value = True

    ChkIsFastener.Value = False

    varTemp1 = InStr(swModel.CustomInfo2("", prpDescription), swModel.CustomInfo2("", prpNumberDocVP))
End If

        ChkEnter.Value = False
    Else

            If varTemp = varTemp1 - 1 Then
                ChkEnter.Value = True
            Else
                ChkEnter.Value = False
            End If
        Else
            ChkEnter.Value = False
        End If
    End If
Else
    ChkEnter.Value = False
End If
mConfig_Click = 1

' Количество в СП

    If mConfig = 1 Then
        If swModel.CustomInfo2(sConfigName, "UNIT_OF_MEASURE") = prpQuantity Then

        End If

        Else
            ChkSP.Value = False

    ok = swModel.AddLightSource("LD2", 4, "Направленный 2")

    doubleTemp1 = Sqr(doubleTemp ^ 2 + 1) * Tan(50 * 3.1416 / 180)

Function Convert(ConvertString As String) As String
Convert = Replace(ConvertString, "|", Chr$(10))
End Function

Function ReConvert(ConvertString As String) As String
ConvertString = Replace(ConvertString, Chr$(10), "|")
ReConvert = Replace(ConvertString, Chr$(13), "")
End Function
