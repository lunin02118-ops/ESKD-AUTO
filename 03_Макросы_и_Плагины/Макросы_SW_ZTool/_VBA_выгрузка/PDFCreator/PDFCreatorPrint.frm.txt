Private Sub Orientation() 'Анализ формата чертежа

vSheetProps = swSheet.GetProperties

If vSheetProps(5) <= vSheetProps(6) Then
    ps.Orientation = swPageSetupOrient_Portrait
    ok = swApp.SetUserPreferenceIntegerValue(swPageSetupPrinterOrientation, swPageSetupOrient_Portrait)
    Debug.Print "ok=", ok
    Select Case vSheetProps(5)
    Case Is <= 0.21
        ps.PrinterPaperSize = 9 ' A4
    Case Is > 0.21
        ps.PrinterPaperSize = 8 ' A3
    End Select
    Select Case vSheetProps(5)
    Case 0.297
        ps.PrinterPaperSize = 8 ' A3
    Case 0.42
        ps.PrinterPaperSize = 66 ' A2
    Case 0.594
        ps.PrinterPaperSize = 145 ' A1
    Case Is >= 0.841
        ps.PrinterPaperSize = 144 ' A0
    End Select
Else
    ps.Orientation = swPageSetupOrient_Landscape
    ok = swApp.SetUserPreferenceIntegerValue(swPageSetupPrinterOrientation, swPageSetupOrient_Landscape)
    Debug.Print "ok=", ok
    Select Case vSheetProps(5) ' Ширина листа
    Case Is <= 0.297
        ps.PrinterPaperSize = 9 ' A4
    Case Is > 0.297
        ps.PrinterPaperSize = 9 ' A3
    End Select
    Select Case vSheetProps(5)
    Case 0.42
        ps.PrinterPaperSize = 8 ' A3
    Case 0.594
        ps.PrinterPaperSize = 66 ' A2
    Case 0.841
        ps.PrinterPaperSize = 145 ' A1
    Case Is >= 1.189
        ps.PrinterPaperSize = 144 ' A0
    End Select
End If

End Sub
