Private Sub ChkScale_Click()
If MChkScale = 0 Then
    ' Узнаем имя активного листа
    Set swSheet = swDraw.GetCurrentSheet
    strActiveSheetName = swSheet.GetName
    ' Активируем первый лист
    vSheetNames = swDraw.GetSheetNames
    ok = swDraw.ActivateSheet(vSheetNames(0))
    Set swSheet = swDraw.GetCurrentSheet
    strSheetFormatName = swSheet.GetSheetFormatName()
    strTemp = "Scale@" & strSheetFormatName
    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
    If ok = True Then
        Set swNote = swSelMgr.GetSelectedObject2(1)
        If ChkScale.Value = True Then
            swNote.SetText "-"
        Else
            swNote.SetText "$PRP:" & Chr$(34) & "SW-Sheet Scale" & Chr$(34)
        End If
    End If
    
    ' Возвращение активного листа
    ok = swDraw.ActivateSheet(strActiveSheetName)
End If
End Sub

Public Sub UserForm_Activate()

Source = swApp.GetCurrentMacroPathName             ' Get macro path+filename
Source1 = Left$(Source, Len(Source) - 3) + "ini"    ' Путь к ini файлу
Source2 = Left$(Source1, Len(Source1) - 15) & "Master\Master.swp" ' Путь к макросу Мастер
Source3 = Left$(Source1, Len(Source1) - 15) & "MProp\MProp.swp" ' Путь к макросу MProp

Source5 = Left$(Source, Len(Source) - 15) & "SpecEditor\MyProperties_1.ini" ' Путь к списку свойств

Source7 = Left$(Source, Len(Source) - 15) & "SpecEditor\MyProperties_2.ini" ' Путь к ini файлу макроса MyProperties
Source8 = Left$(Source, Len(Source) - 15) & "SpecEditor\MyStandard.sldstd" ' Путь к файлу стандарта
Source9 = Left$(Source1, Len(Source1) - 15) & "Master\Master.ini" ' Путь к ini-файлу макроса Мастер
Source10 = Left$(Source, Len(Source) - 15) & "SpecEditor\LRI.slddrt"    ' Шаблон ЛРИ

' Чтение ini файла
Open Source1 For Input As #1
Line Input #1, strTemp
Line Input #1, strTemp
MIni = CInt(Left$(strTemp, 1))
Line Input #1, strTemp
MIniRoug = CInt(Left$(strTemp, 1))
Line Input #1, strTemp
MIniLRI = CInt(Left$(strTemp, 1))
Line Input #1, strTemp
MIniRev = CInt(Left$(strTemp, 1))
Close #1

' Чтение ini файла макроса Мастер
Open Source9 For Input As #1
Line Input #1, strTemp
Line Input #1, strTemp
Line Input #1, strTemp
Line Input #1, MIniMaster
Close #1

MyProperties
If prpTestStandard = 1 Then
    MyStandard
End If

Set swSelMgr = swDraw.SelectionManager

' Узнаем имя активного листа
Set swSheet = swDraw.GetCurrentSheet
strActiveSheetName = swSheet.GetName

' Активируем первый лист
vSheetNames = swDraw.GetSheetNames
ok = swDraw.ActivateSheet(vSheetNames(0))
Set swSheet = swDraw.GetCurrentSheet

'  Получаем параметры модели
MView = 1
Set swView = swDraw.GetFirstView
m = 0
If swSheet.CustomPropertyView = "По умолчанию" Or swSheet.CustomPropertyView = "Default" Then
    Set swView = swView.GetNextView ' Получаем первый вид
Else
    Do Until swView Is Nothing
        If swView.GetName2 = swSheet.CustomPropertyView Then
            m = 1
            Exit Do
        End If
        Set swView = swView.GetNextView
    Loop
    If m = 0 Then
        Set swView = swDraw.GetFirstView
        Set swView = swView.GetNextView
        ' ЕСКД WP-3.5 (S-8): вид пуст или не найден — первый вид листа, без окна
    End If
End If
If swView Is Nothing Then
    MView = 0
Else
    If swView.ReferencedDocument Is Nothing Then
        MView = 0
    Else
        sConfigName = swView.ReferencedConfiguration ' Имя конфигурации вида
        Set swModel = swView.ReferencedDocument
        vModelViewNames = swModel.GetModelViewNames
    End If
End If

' Считываем масштаб
vSheetProps = swSheet.GetProperties
Numerator = vSheetProps(2)
Denominator = vSheetProps(3)
If Numerator < 1 Then
    DTemp = 1 / Numerator
    Numerator = Numerator * DTemp
    Denominator = Denominator * DTemp
End If
If Denominator < 1 Then
    DTemp = 1 / Denominator
    Numerator = Numerator * DTemp
    Denominator = Denominator * DTemp
End If
Debug.Print Numerator, Denominator
' Определяем системный разделитель
If swApp.GetUserPreferenceToggle(swUseSystemSeparatorForDims) Then
    strTemp = swApp.GetUserPreferenceStringValue(swSeparatorCharacterForDims)
Else
    strTemp1 = swApp.GetUserPreferenceStringValue(swSeparatorCharacterForDims)
    swApp.SetUserPreferenceToggle swUseSystemSeparatorForDims, True
    strTemp = swApp.GetUserPreferenceStringValue(swSeparatorCharacterForDims)
    swApp.SetUserPreferenceToggle swUseSystemSeparatorForDims, False
    ok = swApp.SetUserPreferenceStringValue(swSeparatorCharacterForDims, strTemp1)
End If
If strTemp = "." Then
    m = 0
ElseIf strTemp = "," Then
    m = 1
Else
    swApp.SendMsgToUser ("Устанавите системным разделителем точку или запятую")
    CmdCancel_Click
    Exit Sub
End If

' Заполнение списка "Масштаб"
CboScale.Clear
CboScale.AddItem "1:10"
CboScale.AddItem "1:5"
CboScale.AddItem "1:4"
If m = 0 Then
    CboScale.AddItem "1:2.5"
Else
    CboScale.AddItem "1:2,5"
End If
CboScale.AddItem "1:2"
CboScale.AddItem "1:1"
CboScale.AddItem "2:1"
If m = 0 Then
    CboScale.AddItem "2.5:1"
Else
    CboScale.AddItem "2,5:1"
End If
CboScale.AddItem "4:1"
CboScale.AddItem "5:1"
CboScale.AddItem "10:1"
m1 = 0
For i = 0 To CboScale.ListCount - 1
    If CboScale.List(i) = CStr(Numerator) & ":" & CStr(Denominator) Then
        m1 = 1
        Exit For
    End If
Next i
If m1 = 0 Then
    CmdAllScale_Click
Else
    CboScale.Value = CStr(Numerator) & ":" & CStr(Denominator)
End If

' Проверяем флажок ChkScale
MChkScale = 1
ChkScale.Value = False
strSheetFormatName = swSheet.GetSheetFormatName()
strTemp = "Scale@" & strSheetFormatName
ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
If ok = True Then
    Set swNote = swSelMgr.GetSelectedObject2(1)
    If swNote.GetText = "-" Then
        ChkScale.Value = True
    End If
End If
MChkScale = 0

' Проверка редакции чертежа
strTemp = swDraw.CustomInfo2("", "Revision")
If strTemp = "" Then
    LblGetRev.Caption = "нет"
    SpinButton1.Value = 1
Else
    LblGetRev.Caption = strTemp
    SpinButton1.Value = CInt(strTemp) + 1
End If
' № докум.
strTemp = "Revision3@" & strSheetFormatName
ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
If ok = True Then
    Set swNote = swSelMgr.GetSelectedObject2(1)
    TxtDoc.Value = swNote.GetText()
End If

Tests
' ЕСКД Д-64: открытие формы чертёж не меняет — листы и «Листов» правит «Исправить оформление чертежа»

' Возвращение активного листа
ok = swDraw.ActivateSheet(strActiveSheetName)

If prpLeftTopCorner = 1 Then ' Окно макроса в левом верхнем углу
    FrmDProp.Left = 10
    FrmDProp.Top = 10
End If
If prpTopAll = 1 Then ' Окно макроса поверх всех
    KeepFormOnTop
End If
End Sub

Private Sub CmdReload_Click() ' Перезагрузка форматки

Private Sub CmdRev_Click() ' Изменить версию
' Узнаем имя активного листа
Set swSheet = swDraw.GetCurrentSheet
strActiveSheetName = swSheet.GetName

' Изм.
If LblSetRev.Caption = "нет" Then
    ok = swDraw.AddCustomInfo3("", "Revision", 30, "")
    swDraw.CustomInfo2("", "Revision") = ""
Else
    ok = swDraw.AddCustomInfo3("", "Revision", 30, LblSetRev.Caption)
    swDraw.CustomInfo2("", "Revision") = LblSetRev.Caption
End If

' Определяем дату
'MyDate = Date
'MyDay = Day(MyDate)
'MyMonth = Month(MyDate)
'MyYear = Year(MyDate)
'strData = CStr(MyDay) & "." & CStr(MyMonth) & "." & Right$(CStr(MyYear), 2)
' Устанавливаем Лист и Дату
vSheetNames = swDraw.GetSheetNames
For i = 0 To UBound(vSheetNames)
    ok = swDraw.ActivateSheet(vSheetNames(i))
    Set swSheet = swDraw.GetCurrentSheet
    strSheetFormatName = swSheet.GetSheetFormatName()
    ' Лист
    strTemp = "Revision2@" & strSheetFormatName
    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
    If ok = True Then
        ' Определяем и устанавливаем Лист
        If LblSetRev.Caption = "нет" Then
            strTemp = " "
        Else
            strTemp = "Зам."
        End If
        Set swNote = swSelMgr.GetSelectedObject2(1)
        If swNote.GetText = "Нов." And MAdd = 1 Then
        Else
            swNote.SetText strTemp
        End If
    End If
    ' № докум.
    strTemp = "Revision3@" & strSheetFormatName
    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
    If ok = True Then
        Set swNote = swSelMgr.GetSelectedObject2(1)
        swNote.SetText TxtDoc.Value
    End If
    ' Дата
    strTemp = "Revision4@" & strSheetFormatName
    ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
    If ok = True Then
        strData = Format(Now, "dd.mm.yy")
        ' Определяем и устанавливаем дату
        If LblSetRev.Caption = "нет" Then
            strTemp = " "
        Else
            strTemp = strData
        End If
        Set swNote = swSelMgr.GetSelectedObject2(1)
        swNote.SetText strTemp
    End If
Next i

' Возвращение активного листа
ok = swDraw.ActivateSheet(strActiveSheetName)

' Перестраиваем чертеж
swDraw.Rebuild (swRebuildAll)
ok = swDraw.GraphicsRedraw2()
swDraw.ClearSelection2 True

' Изменение отображения редакции чертежа
strTemp = swDraw.CustomInfo2("", "Revision")
If strTemp = "" Then
    LblGetRev.Caption = "нет"
    SpinButton1.Value = 1
Else
    LblGetRev.Caption = strTemp
    SpinButton1.Value = CInt(strTemp) + 1
End If

End Sub

End Sub

Private Sub SheetsControl()

' Переименовываем листы и выставляем масштаб
vSheetNames = swDraw.GetSheetNames
j = 0
m = 0
k = 0
k1 = 0
'Debug.Print "UBound(vSheetNames)", UBound(vSheetNames)
For i = 0 To UBound(vSheetNames)
    ' Проверка имени листа
    If Left$(vSheetNames(i), 3) = "LRI" Then
        m = 1
        k = i
        j = j + 1
    End If
    If Left$(vSheetNames(i), 3) = "DRW" Or Left$(vSheetNames(i), 4) = "Лист" Or Left$(vSheetNames(i), 5) = "Sheet" Then
        j = j + 1
        k1 = i
        ok = swDraw.ActivateSheet(vSheetNames(i))
        Set swSheet = swDraw.GetCurrentSheet
        strTemp = "DRW" & CStr(j)
        vSheetProps = swSheet.GetProperties
        ok = swDraw.SetupSheet4(strTemp, vSheetProps(0), vSheetProps(1), Numerator, Denominator, vSheetProps(4), swSheet.GetTemplateName, vSheetProps(5), vSheetProps(6), swSheet.CustomPropertyView)
    End If
Next i

' Вставляем ЛРИ
If MIniLRI = 1 And j > 2 And m = 0 Then ' вставляем ЛРИ
    strTemp = "LRI"
    vRetval = swDraw.NewSheet3(strTemp, swDwgPapersUserDefined, swDwgTemplateCustom, 1#, 1#, True, Source10, 0.21, 0.297, "По умолчанию")
    Set swSheet = swDraw.GetCurrentSheet
    swSheet.SheetFormatVisible = True
    If MView = 1 Then ' Вставляем вид
        Set swView = swDraw.CreateDrawViewFromModelView3(swModel.GetPathName, vModelViewNames(0), -0.1, 0, 0)
        swView.ReferencedConfiguration = sConfigName
        swDraw.SuppressView
        swDraw.ForceRebuild3 (True)
    End If
    j = j + 1
    ' Тасуем листы
    vSheetNames = swDraw.GetSheetNames
    For i = UBound(vSheetNames) To k1 + 2 Step -1
        vSheetNames(i) = vSheetNames(i - 1)
    Next i
    vSheetNames(k1 + 1) = strTemp
    ok = swDraw.ReorderSheets(vSheetNames)
ElseIf MIniLRI = 1 And j < 4 And m = 1 Then   ' удаляем ЛРИ
    strMsg = "Удалить ЛРИ?"
    lRetval = swApp.SendMsgToUser2(strMsg, swMbWarning, swMbYesNo)
    If lRetval = swMbHitYes Then
        ok = swDraw.ActivateSheet(vSheetNames(k))
        Set swSheet = swDraw.GetCurrentSheet
        DeleteOption = 3
        ok = swDraw.Extension.SelectByID2(swSheet.GetName, "SHEET", 0, 0, 0, False, 0, Nothing, 0)
        ok = swDraw.Extension.DeleteSelection2(DeleteOption)
        j = j - 1
    End If
End If

' Заносим количество листов
vSheetNames = swDraw.GetSheetNames
ok = swDraw.ActivateSheet(vSheetNames(0))
Set swSheet = swDraw.GetCurrentSheet
strSheetFormatName = swSheet.GetSheetFormatName()
strTemp = "Sheet1@" & strSheetFormatName
ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
If ok = True Then
    If j > 1 Then ' Листов больше одного
        strTemp = "Лист 1"
    Else
        strTemp = "Лист"
    End If
    Set swNote = swSelMgr.GetSelectedObject2(1)
    swNote.SetText strTemp
End If
strTemp = "Sheet2@" & strSheetFormatName
ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
If ok = True Then
    strTemp = "Листов " & j
    Set swNote = swSelMgr.GetSelectedObject2(1)
    swNote.SetText CStr(strTemp)
End If

End Sub

Private Sub Tests()
If prpTestVersion = 1 Or prpTestFormat = 1 Or prpTestStandard = 1 Then
    vSheetNames = swDraw.GetSheetNames
    ReDim sFormatArray(UBound(vSheetNames))
    ' Считываем и проверяем версии форматов всех листов
    k = 0
    k1 = 0
    k2 = 0
    k3 = 0
    sSheetsNames1 = ""
    sSheetsNames2 = ""
    sSheetsNames3 = ""
    sSheetsNames4 = ""
    strMsg = ""
    For i = 0 To UBound(vSheetNames)
        ok = swDraw.ActivateSheet(vSheetNames(i))
        Set swSheet = swDraw.GetCurrentSheet
        strSheetFormatName = swSheet.GetSheetFormatName()
        ' Проверка имени листа
        If Left$(vSheetNames(i), 3) = "DRW" Or Left$(vSheetNames(i), 4) = "Лист" Or Left$(vSheetNames(i), 5) = "Sheet" Or Left$(vSheetNames(i), 2) = "SP" Or Left$(vSheetNames(i), 2) = "VP" Or Left$(vSheetNames(i), 3) = "LRI" Then
            If prpTestVersion = 1 Then
                ' Проверка версии форматки
                strTemp = "Version@" & strSheetFormatName
                ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
                If ok = True Then
                    Set swNote = swSelMgr.GetSelectedObject2(1)
                    strTemp = swNote.GetText()
                    ' Проверка версии
                    If Right$(strTemp, 1) <> "3" Then
                        k1 = k1 + 1
                        sSheetsNames1 = sSheetsNames1 & vSheetNames(i) & ", "
                    End If
                Else
                    k1 = k1 + 1
                    sSheetsNames1 = sSheetsNames1 & vSheetNames(i) & ", "
                End If
            End If
            If prpTestStandard = 1 Then
                ' Проверка оформления
                mEx1 = 0
                mEx2 = 0
                Set swView = swDraw.GetFirstView ' Проверяем шрифты форматки
                If Not swView Is Nothing Then
                    Set swNote = swView.GetFirstNote()
                    j = 0
                    While j = 0
                        If Not swNote Is Nothing Then
                            Set swAnn = swNote.GetAnnotation()
                            If Not swAnn.OwnerType = swAnnotationOwner_DrawingTemplate Then ' Исключаем объекты форматки
                                Set swNote = swNote.GetNext
                            Else
                                j = 1
                            End If
                        Else
                            j = 1
                        End If
                    Wend
                    If Not swNote Is Nothing Then
                        Set swAnn = swNote.GetAnnotation()
                        Set swTextFormat = swAnn.GetTextFormat(0)
                        If swTextFormat.TypeFaceName <> stdFontName Then
                            mEx1 = 1
                        End If
                        If swTextFormat.Italic = True And stdFontItalic = 0 Then
                            mEx1 = 1
                        ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                            mEx1 = 1
                        End If
                        If swTextFormat.Bold = True And stdFontBold = 0 Then
                            mEx1 = 1
                        ElseIf swTextFormat.Bold = False And stdFontBold = 1 Then
                            mEx1 = 1
                        End If
                    Else
                        mEx1 = 1
                    End If
                Else
                    mEx1 = 1
                End If
                Set swView = swView.GetNextView ' Проверяем шрифты чертежа
                If Not swView Is Nothing Then
                    Set swNote = swView.GetFirstNote()
                    If Not swNote Is Nothing Then
                        Set swAnn = swNote.GetAnnotation()
                        Set swTextFormat = swAnn.GetTextFormat(0)
                        If swTextFormat.TypeFaceName <> stdFontName Then
                            mEx2 = 1
                        End If
                        If swTextFormat.Italic = True And stdFontItalic = 0 Then
                            mEx2 = 1
                        ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                            mEx2 = 1
                        End If
                        If swTextFormat.Bold = True And stdFontBold = 0 Then
                            mEx2 = 1
                        ElseIf swTextFormat.Bold = False And stdFontBold = 1 Then
                            mEx2 = 1
                        End If
                    End If
                End If
                Set swModExt = swDraw.Extension
                Set swTextFormat = swModExt.GetUserPreferenceTextFormat(swDetailingDimensionTextFormat, swDetailingDimension)
                If swTextFormat.TypeFaceName <> stdFontName Or swTextFormat.CharHeight <> stdFontSize / 1000 Then
                    mEx2 = 1
                End If
                If swTextFormat.Italic = True And stdFontItalic = 0 Then
                    mEx2 = 1
                ElseIf swTextFormat.Italic = False And stdFontItalic = 1 Then
                    mEx2 = 1
                End If
                If swTextFormat.Bold = True And stdFontBold = 0 Then
                    mEx2 = 1
                ElseIf swTextFormat.Bold = False And stdFontBold = 1 Then
                    mEx2 = 1
                End If
                If mEx1 = 1 Then
                    k3 = k3 + 1
                    sSheetsNames4 = sSheetsNames4 & vSheetNames(i) & ", "
                End If
                If mEx2 = 1 Then
                    k2 = k2 + 1
                    sSheetsNames2 = sSheetsNames2 & vSheetNames(i) & ", "
                End If
            End If
        End If
        If prpTestFormat = 1 Then
            If Left$(vSheetNames(i), 3) = "DRW" Or Left$(vSheetNames(i), 4) = "Лист" Or Left$(vSheetNames(i), 5) = "Sheet" Or Left$(vSheetNames(i), 3) = "LRI" Then
                ' Проверка имени формата
                strTemp = "Format@" & strSheetFormatName
                ok = swDraw.Extension.SelectByID2(strTemp, "NOTE", 0, 0, 0, False, 0, Nothing, 0)
                If ok = True Then
                    Set swNote = swSelMgr.GetSelectedObject2(1)
                    strTemp = swNote.GetText()
                    ' Проверка корректности и длины записи
                    If Len(Trim(strTemp)) > 8 And Left$(strTemp, 6) = "Формат" Then
                    Else
                        k = k + 1
                        sSheetsNames3 = sSheetsNames3 & vSheetNames(i) & ", "
                    End If
                Else
                    k = k + 1
                    sSheetsNames3 = sSheetsNames3 & vSheetNames(i) & ", "
                End If
            End If
        End If
    Next i
    If k > 0 Or k1 > 0 Or k2 > 0 Or k3 > 0 Then  ' Предупреждение об ошибках
        If k1 > 0 Then
            sSheetsNames1 = Left$(sSheetsNames1, Len(sSheetsNames1) - 2)
            strMsg = "Лист(ы) " & sSheetsNames1 & " имеют нестандартную или устаревшую основную надпись."
        End If
        If k2 > 0 Then
            sSheetsNames2 = Left$(sSheetsNames2, Len(sSheetsNames2) - 2)
            strMsg = strMsg & Chr$(10) & "Лист(ы) " & sSheetsNames2 & " имеют ошибки оформления чертежа."
        End If
        If k3 > 0 Then
            sSheetsNames4 = Left$(sSheetsNames4, Len(sSheetsNames4) - 2)
            strMsg = strMsg & Chr$(10) & "Лист(ы) " & sSheetsNames4 & " имеют ошибки оформления основной надписи."
        End If
        If k > 0 Then
            sSheetsNames3 = Left$(sSheetsNames3, Len(sSheetsNames3) - 2)
            strMsg = strMsg & Chr$(10) & "Лист(ы) " & sSheetsNames3 & " не позволяют определить формат."
        End If
        strMsg = strMsg & Chr$(10) & "Исправить?"
        lRetval = swApp.SendMsgToUser2(strMsg, swMbWarning, swMbYesNo)
        If lRetval = swMbHitYes Then
            Set swModExt = swDraw.Extension
            ok = swModExt.LoadDraftingStandard(Source8)
            If MIni = 1 Then
                Add ' Доп. оформление
            End If
            Impr ' Перегружаем все форматки чертежа
        End If
    End If
End If
End Sub

Private Sub Impr() ' Перезагрузка форматок чертежа
    MImpr = 1
    For j = 0 To UBound(vSheetNames)
        If Left$(vSheetNames(j), 3) = "DRW" Or Left$(vSheetNames(j), 4) = "Лист" Or Left$(vSheetNames(j), 5) = "Sheet" Or Left$(vSheetNames(j), 2) = "SP" Or Left$(vSheetNames(j), 2) = "VP" Or Left$(vSheetNames(j), 3) = "LRI" Then
            varTemp = InStr(sSheetsNames1, vSheetNames(j)) ' Проверяем имя
            varTemp1 = InStr(sSheetsNames3, vSheetNames(j)) ' Проверяем имя
            varTemp2 = InStr(sSheetsNames4, vSheetNames(j)) ' Проверяем имя
            If varTemp > 0 Or varTemp1 > 0 Or varTemp2 > 0 Then ' Значение присутствует
                ok = swDraw.ActivateSheet(vSheetNames(j))
                Set swSheet = swDraw.GetCurrentSheet
                CmdReload_Click
                varTemp = InStr(sSheetsNames2, vSheetNames(j)) ' Проверяем имя
                If varTemp > 0 Then ' Значение присутствует
                    AnnDef
                End If
            Else
                varTemp = InStr(sSheetsNames2, vSheetNames(j)) ' Проверяем имя
                If varTemp > 0 Then ' Значение присутствует
                    AnnDef
                End If
            End If
        End If
    Next j
    Tests
    MImpr = 0
End Sub

Private Sub AnnDef() ' Выставляем все по умолчанию
'Dim vvv As Variant
' Поиск объектов со шрифтом не по умолчанию
Set swView = swDraw.GetFirstView
Do While Not swView Is Nothing
    Set swAnn = swView.GetFirstAnnotation3
    Do While Not swAnn Is Nothing
        If Not swAnn.OwnerType = swAnnotationOwner_DrawingTemplate Then ' Исключаем объекты форматки
            If swAnn.GetUseDocTextFormat(0) = False Then ' Шрифт не по умолчанию
                Set swTextFormat = swAnn.GetTextFormat(0)
                ok = swAnn.SetTextFormat(0, True, swTextFormat) ' Выставляем по умолчанию
            End If
            If swAnn.GetType = swSFSymbol Then ' Значек шероховатости
                Set swAnnSFSymbol = swAnn.GetSpecificAnnotation
                strTemp = swAnnSFSymbol.GetText(swSFSymbolRoughnessSpacing)
                If strTemp <> "" Then
                    varTemp = InStrRev(strTemp, ">")
                    If varTemp > 0 Then
                        strTemp = Right$(strTemp, Len(strTemp) - varTemp)
                    End If
                Else ' Текст над значком
                    strTemp = swAnnSFSymbol.GetText(swSFSymbolMinimumRoughness)
                    ok = swAnnSFSymbol.SetText(swSFSymbolMinimumRoughness, "")
                End If
                strTemp = "<FONT size=" & LTrim(CStr(stdFontSize)) & ">" & strTemp
                ok = swAnnSFSymbol.SetText(swSFSymbolRoughnessSpacing, strTemp)
            End If
        End If
        'Debug.Print "Ann Type=", swAnn.GetType
        'Debug.Print "Ann Name=", swAnn.GetName
        'vvv = swAnn.GetPosition
        'Debug.Print "Ann Pos=", vvv(0), vvv(1)
        Set swAnn = swAnn.GetNext3
    Loop
    Set swView = swView.GetNextView
Loop
End Sub
