Private Sub CmdPRPName_Click() ' Запускаем MyProperties
ok = swApp.RunMacro(Source6, "MyProperties_run", "main")

Private Sub UserForm_Activate()
HWNDActiveWindow
' Чтение ini файла
Open Source1 For Input As #1
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkIns.Value = True
    Else
        ChkIns.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkAdd.Value = True
    Else
        ChkAdd.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkRoug.Value = True
    Else
        ChkRoug.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkLRI.Value = True
    Else
        ChkLRI.Value = False
    End If
    Line Input #1, strTemp
    If strTemp = "1" Then
        ChkRev.Value = True
    Else
        ChkRev.Value = False
    End If
Close #1
If prpLeftTopCorner = 1 Then ' Окно макроса в левом верхнем углу
    FrmDProp_Pref.Left = 10
    FrmDProp_Pref.Top = 10
End If
If prpTopAll = 1 Then ' Окно макроса поверх всех
    KeepFormOnTop
End If
End Sub

Private Sub CmdSave_Click()
' Запись в ini
Open Source1 For Output As #1
    If ChkIns.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkAdd.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkRoug.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkLRI.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
    If ChkRev.Value = True Then
        Print #1, "1"
    Else
        Print #1, "0"
    End If
Close #1
Hide
FrmDProp.UserForm_Activate
End Sub
